using Microsoft.AspNetCore.Mvc;

public class HomeController : Controller
{
    public IActionResult Index() => View();

    [HttpPost]
    public IActionResult Index(double a, double b, string op)
    {
        double result = op switch
        {
            "Add" => a + b,
            "Subtract" => a - b,
            "Multiply" => a * b,
            "Divide" => b != 0 ? a / b : 0,
            _ => 0
        };
        ViewBag.Result = result;
        return View();
    }
}

// service-worker.js
const CACHE_NAME = "college-cache-v1";
const FILES = ["./", "./index.html", "./offline.html"];

self.addEventListener("install", event => {
  event.waitUntil(
    caches.open(CACHE_NAME).then(cache => cache.addAll(FILES))
  );
});

self.addEventListener("fetch", event => {
  event.respondWith(
    fetch(event.request).catch(() => caches.match(event.request)
      .then(response => response || caches.match("./offline.html")))
  );
});

  • Service workers normally require HTTPS, with localhost allowed for development.

let students = JSON.parse(localStorage.getItem("students") || "[]");

function showStudents() {
  const list = document.getElementById("list");
  list.innerHTML = "";
  students.forEach((s, i) => {
    list.innerHTML += `<li>${s.name} - ${s.course}
      <button onclick="removeStudent(${i})">Delete</button></li>`;
  });
}

function addStudent() {
  students.push({
    name: document.getElementById("name").value,
    course: document.getElementById("course").value
  });
  localStorage.setItem("students", JSON.stringify(students));
  showStudents();
}

function removeStudent(i) {
  students.splice(i, 1);
  localStorage.setItem("students", JSON.stringify(students));
  showStudents();
}

showStudents();

if ("serviceWorker" in navigator)
  navigator.serviceWorker.register("service-worker.js");
</script>
</body>
</html>

ASP.NET Core Web API + EF Core

4A. CRUD API — Controller

Steps
  1. Run the API.
  2. POST /api/students with JSON to create.
  3. GET /api/students to read.
  4. PUT /api/students/1 to update.
  5. DELETE /api/students/1 to delete.
  6. Use Postman Body → raw → JSON for POST/PUT.
Code
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

[ApiController]
[Route("api/[controller]")]
public class StudentsController : ControllerBase
{
    private readonly StudentDbContext db;
    public StudentsController(StudentDbContext db) => this.db = db;

    [HttpGet]
    public async Task<ActionResult<List<Student>>> Get()
        => await db.Students.ToListAsync();

    [HttpGet("{id}")]
    public async Task<ActionResult<Student>> Get(int id)
    {
        var s = await db.Students.FindAsync(id);
        return s == null ? NotFound() : s;
    }

    [HttpPost]
    public async Task<ActionResult<Student>> Post(Student s)
    {
        db.Students.Add(s);
        await db.SaveChangesAsync();
        return Ok(s);
    }

    [HttpPut("{id}")]
    public async Task<IActionResult> Put(int id, Student s)
    {
        var old = await db.Students.FindAsync(id);
        if (old == null) return NotFound();
        old.Name = s.Name;
        old.Course = s.Course;
        await db.SaveChangesAsync();
        return Ok(old);
    }

    [HttpDelete("{id}")]
    public async Task<IActionResult> Delete(int id)
    {
        var s = await db.Students.FindAsync(id);
        if (s == null) return NotFound();
        db.Students.Remove(s);
        await db.SaveChangesAsync();
        return NoContent();
    }
}
Viva Quick Points
  • CRUD = Create, Read, Update, Delete.
  • POST = Create; GET = Read; PUT = Update; DELETE = Delete.
  • SaveChangesAsync saves EF Core changes.

4B. Factorial API + Postman

Steps

  2. Add FactorialController.cs.
  3. Run the project and note the localhost port.

  5. Request: https://localhost:PORT/api/factorial/5

Code
using Microsoft.AspNetCore.Mvc;

[ApiController]
[Route("api/[controller]")]
public class FactorialController : ControllerBase
{
    [HttpGet("{number}")]
    public IActionResult GetFactorial(int number)
    {
        if (number < 0)
            return BadRequest("Number must be non-negative.");

        long factorial = 1;
        for (int i = 1; i <= number; i++)
            factorial *= i;

        return Ok(new { Number = number, Factorial = factorial });
    }
}
Viva Quick Points
  • Postman tests HTTP APIs.
  • GET retrieves a resource/result.
  • Factorial of 5 = 5×4×3×2×1 = 120.
  • 0! = 1.
